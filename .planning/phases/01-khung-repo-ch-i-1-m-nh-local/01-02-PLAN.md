export type TapEvent = { readonly value: number; readonly t: number };
export type GameEvent = { type: 'start'; t: number } | { type: 'tap'; value: number; t: number };
export type GameStatus = 'ready' | 'playing' | 'finished';
export interface GameState {
  readonly status: GameStatus;
  readonly seed: string;
  readonly size: BoardSize;
  readonly total: number;               // size * size
  readonly board: readonly number[];    // generateBoard(seed, size)
  readonly next: number;                // next expected number, starts at 1

}

export function gameReducer(state: GameState, event: GameEvent): GameState;
